```js
function getData(annee, etab) {
  let d = rawData.filter(r => r.Annee === annee);
  if (etab !== "Tout le Québec") d = d.filter(r => r.NomEtablissement === etab);
  const agg = {};
  for (const r of d) { const k = `${r.Ligne}-${r.Colonne}`; agg[k] = (agg[k]||0)+(r.Chiffre||0); }
  return agg;
}
const AGE = {1:'0-4 ans', 2:'5-11 ans', 3:'12-17 ans', 4:'18-21 ans', 5:'22-44 ans', 6:'45-64 ans', 7:'65-74 ans', 8:'75 ans+', 9:'Total'};
function v(data, l, c) { return Math.round(data[`${l}-${c}`] || 0); }
function fmtD(n) { return n > 0 ? n.toFixed(1) : "—"; }
function fmt(n) { return n.toLocaleString('fr-CA'); }
const data = getData(anneeSelect, etabSelect);
```

const trendFlat = meta.annees.flatMap(annee => {
  const d = (() => {
    let rows = rawData.filter(r => r.Annee === annee);
    if (etabSelect !== "Tout le Québec") rows = rows.filter(r => r.NomEtablissement === etabSelect);
    const agg = {};
    for (const r of rows) { const k = `${r.Ligne}-${r.Colonne}`; agg[k] = (agg[k]||0)+(r.Chiffre||0); }
    return agg;
  })();
  return [
    {annee, n: v(d,1,9), serie: "DI"},
    {annee, n: v(d,4,9), serie: "TSA"},
    {annee, n: v(d,7,9), serie: "Total DITSA"},
  ];
}).filter(d => d.n > 0);

Plot.plot({
