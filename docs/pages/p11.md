```js
function getData(annee, etab) {
  let d = rawData.filter(r => r.Annee === annee);
  if (etab !== "Tout le Québec") d = d.filter(r => r.NomEtablissement === etab);
  const agg = {};
  for (const r of d) {
    const key = `${r.Ligne}-${r.Colonne}`;
    agg[key] = (agg[key] || 0) + (r.Chiffre || 0);
  }
  return agg;
}
const AGE = {1:'0-4 ans', 2:'5-11 ans', 3:'12-17 ans', 4:'18-21 ans', 5:'22-44 ans', 6:'45-64 ans', 7:'65-74 ans', 8:'75 ans+', 9:'Total'};
function v(data, l, c) { return Math.round(data[`${l}-${c}`] || 0); }
function fmt(n) { return n.toLocaleString('fr-CA'); }
const data = getData(anneeSelect, etabSelect);
```

## Assistance résidentielle continue — DI (s-c/a 7041)

```js
const RRAC_DI = [
  {ligne: 1, label: "Usagers desservis durant l'année"},
  {ligne: 2, label: "Départs durant l'année"},
  {ligne: 3, label: "Nombre de jours-présence"},
];
const tableRRAC_DI = RRAC_DI.map(r => ({
  "Indicateur": r.label,
  ...Object.fromEntries([1,2,3,4,5,6,7,8].map(c => [AGE[c], fmt(v(data, r.ligne, c))])),
```
