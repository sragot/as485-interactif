```js
function getData(annee, etab) {
  let d = rawData.filter(r => r.Annee === annee);
  if (etab !== "Tout le Québec") d = d.filter(r => r.NomEtablissement === etab);
  const agg = {};
  for (const r of d) {
    const key = `${r.Ligne}-${r.Colonne}`;
    agg[key] = (agg[key] || 0) + (r.Chiffre || 0);
  }
  return agg;
}
function v(data, l, c) { return Math.round(data[`${l}-${c}`] || 0); }
function fmt(n) { return n.toLocaleString('fr-CA'); }
const data = getData(anneeSelect, etabSelect);
```

## Graphique — Répartition des places autorisées

```js
const chartData = TYPES.map(t => ({
  type: t.label,
  places: v(data, 40, t.col),
})).filter(d => d.places > 0);

Plot.plot({
  title: `Places autorisées — ${etabSelect} — ${anneeSelect}`,
  x: {label: "Places"},
  y: {label: null},
  marginLeft: 180,
  marks: [
    Plot.barX(chartData, {x: "places", y: "type", fill: "#378ADD", sort: {y: "-x"}}),
```
