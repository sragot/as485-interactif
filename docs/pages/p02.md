```js
function getData(annee, etab) {
  let d = rawData.filter(r => r.Annee === annee);
  if (etab !== "Tout le Québec") d = d.filter(r => r.NomEtablissement === etab);
  const agg = {};
  for (const r of d) {
    const key = `${r.Ligne}-${r.Colonne}`;
    agg[key] = (agg[key] || 0) + (r.Chiffre || 0);
  }
  return agg;
}
function v(data, l, c) { return Math.round(data[`${l}-${c}`] || 0); }
function fmt(n) { return n.toLocaleString('fr-CA'); }
const data = getData(anneeSelect, etabSelect);
```

## Résumé — Internats (Ligne 40 = Total)

```js
const summary = [
  {label: "Jours-présence internat (C.3)", val: fmt(v(data, 40, 3))},
  {label: "Nombre de lits dressés au 31 mars (C.4)", val: fmt(v(data, 40, 4))},
  {label: "Taux d'occupation % (C.5)", val: (v(data, 40, 5) / 10).toFixed(1) + " %"},
```

## Évolution historique — Jours-présence internat

```js
const trendData = meta.annees.map(annee => {

    let rows = rawData.filter(r => r.Annee === annee);
    if (etabSelect !== "Tout le Québec") rows = rows.filter(r => r.NomEtablissement === etabSelect);
    const agg = {};

    return agg;
  })();
  return {annee, joursPresence: v(d, 40, 3), lits: v(d, 40, 4)};
}).filter(d => d.joursPresence > 0 || d.lits > 0);
```
