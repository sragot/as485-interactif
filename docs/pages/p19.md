```js
function getData(annee, etab) {
  let d = rawData.filter(r => r.Annee === annee);
  if (etab !== "Tout le Québec") d = d.filter(r => r.NomEtablissement === etab);
  const agg = {};
  for (const r of d) { const k = `${r.Ligne}-${r.Colonne}`; agg[k] = (agg[k]||0)+(r.Chiffre||0); }
  return agg;
}
const AGE = {1:'0-4 ans', 2:'5-11 ans', 3:'12-17 ans', 4:'18-21 ans', 5:'22-44 ans', 6:'45-64 ans', 7:'65-74 ans', 8:'75 ans+', 9:'Total'};
function v(data, l, c) { return Math.round(data[`${l}-${c}`] || 0); }
function fmtD(n) { return n > 0 ? n.toFixed(1) : "—"; }
function fmt(n) { return n.toLocaleString('fr-CA'); }
const data = getData(anneeSelect, etabSelect);
```

```js
const SERVICES = [
  {lignes:[1,2,3], label:"Intégration communautaire (s-c/a 7001)"},
  {lignes:[4,5,6], label:"Intégration au travail (s-c/a 7011, 7024, 7025, 7031)"},
  {lignes:[7,8,9], label:"Intégration résidentielle (s-c/a 5516-7051)"},
  {lignes:[10,11,12], label:"Adaptation/réadaptation à la personne (s-c/a 8052 et 8053)"},
];

for (const svc of SERVICES) {
  const [l1, l2, l3] = svc.lignes;
  const rows = [
    {label: "Nombre de personnes", ligne: l1},
    {label: "Délai moyen (jours)", ligne: l2},
    {label: "Délai médian (jours)", ligne: l3},
  ];
  const tableRows = rows.map(r => ({
    "Indicateur": r.label,
    ...Object.fromEntries([1,2,3,4,5,6,7,8].map(c => [AGE[c], r.ligne === l1 ? fmt(v(data,r.ligne,c)) : fmtD(v(data,r.ligne,c))])),
    "Total": r.ligne === l1 ? fmt(v(data,r.ligne,9)) : fmtD(v(data,r.ligne,9))
  }));
  display(html`<h3 style="font-size:14px; font-weight:500; margin: 1rem 0 0.5rem;">${svc.label}</h3>`);
  display(tableWithTools(Inputs.table(tableRows, {width:"100%"}), tableRows, `p19_l${l1}.csv`));
}
```
