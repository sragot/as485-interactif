```js
// Filter data
function getData(annee, etab) {
  let d = rawData.filter(r => r.Annee === annee);
  if (etab !== "Tout le Québec") {
    d = d.filter(r => r.NomEtablissement === etab);
  }
  // Aggregate by Ligne/Colonne
  const agg = {};
  for (const r of d) {
    const key = `${r.Ligne}-${r.Colonne}`;
    agg[key] = (agg[key] || 0) + (r.Chiffre || 0);
  }
  return agg;
}

const AGE = {1:'0-4 ans', 2:'5-11 ans', 3:'12-17 ans', 4:'18-21 ans', 5:'22-44 ans', 6:'45-64 ans', 7:'65-74 ans', 8:'75 ans+', 9:'Total'};
const DEF = {1:'DI', 2:'TSA', 3:'DI+TSA', 4:'DI+Autre', 5:'TSA+Autre', 6:'DI+TSA+Autre', 7:'Autre', 8:'N/P', 9:'Total'};

function v(data, l, c) { return Math.round(data[`${l}-${c}`] || 0); }
function fmt(n) { return n.toLocaleString('fr-CA'); }

const data = getData(anneeSelect, etabSelect);
```

## Graphique — Clients par groupe d'âge

```js
// Build chart data
const chartData = [];
for (const c of [1,2,3,4,5,6,7,8]) {
  chartData.push({
    age: AGE[c],
    jeunes: v(data, 1, c),
    adultes: v(data, 11, c),
  });
}

Plot.plot({
  title: `Clientèle par groupe d'âge — ${etabSelect} — ${anneeSelect}`,
  x: { label: "Groupe d'âge" },
  y: { label: "Usagers" },
  color: { legend: true },
  marks: [
    Plot.barY(chartData, Plot.groupX({y: "sum"}, {x: "age", y: "jeunes", fill: "#1D9E75", title: d => `0-17 ans\n${d.age}: ${d.jeunes}`})),
```
