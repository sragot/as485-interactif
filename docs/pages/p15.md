```js
function getData(annee, etab) {
  let d = rawData.filter(r => r.Annee === annee);
  if (etab !== "Tout le Québec") d = d.filter(r => r.NomEtablissement === etab);
  const agg = {};
  for (const r of d) { const k = `${r.Ligne}-${r.Colonne}`; agg[k] = (agg[k]||0)+(r.Chiffre||0); }
  return agg;
}
function v(data, l, c) { return Math.round(data[`${l}-${c}`] || 0); }
function fmt(n) { return n.toLocaleString('fr-CA'); }
const data = getData(anneeSelect, etabSelect);
```
