```js
function getData(annee, etab) {
  let d = rawData.filter(r => r.Annee === annee);
  if (etab !== "Tout le Québec") d = d.filter(r => r.NomEtablissement === etab);
  const agg = {};
  for (const r of d) {
    const key = `${r.Ligne}-${r.Colonne}`;
    agg[key] = (agg[key] || 0) + (r.Chiffre || 0);
  }
  return agg;
}

const AGE = {1:'0-4 ans', 2:'5-11 ans', 3:'12-17 ans', 4:'18-21 ans', 5:'22-44 ans', 6:'45-64 ans', 7:'65-74 ans', 8:'75 ans+', 9:'Total'};
function v(data, l, c) { return Math.round(data[`${l}-${c}`] || 0); }
function fmt(n) { return n.toLocaleString('fr-CA'); }

const data = getData(anneeSelect, etabSelect);
```

## Demandes de services reçues, débutées et fermées

```js
const ROWS16 = [
  { ligne: 1, label: "Reçues — Déficience intellectuelle (DI)" },
  { ligne: 2, label: "Reçues — TSA" },
  { ligne: 3, label: "Reçues — TOTAL (L.1 + L.2)" },
  { ligne: 4, label: "Reçues dont services débutés — DI" },
  { ligne: 5, label: "Reçues dont services débutés — TSA" },
  { ligne: 6, label: "Reçues dont services débutés — TOTAL" },
  { ligne: 7, label: "Fermées — DI" },
  { ligne: 8, label: "Fermées — TSA" },
  { ligne: 9, label: "Fermées — TOTAL" },
];

const table16 = ROWS16.map(r => ({
  "Indicateur": r.label,
  ...Object.fromEntries([1,2,3,4,5,6,7,8].map(c => [AGE[c], fmt(v(data, r.ligne, c))])),
  "Total": fmt(v(data, r.ligne, 9)),
}));
view(tableWithTools(Inputs.table(table16, {width: "100%"}), table16, "p16_demandes.csv"));
```
