```js
function getData(raw, annee, etab) {
  let d = raw.filter(r => r.Annee === annee);
  if (etab !== "Tout le Québec") d = d.filter(r => r.NomEtablissement === etab);
  const agg = {};
  for (const r of d) {
    const key = `${r.Ligne}-${r.Colonne}`;
    agg[key] = (agg[key] || 0) + (r.Chiffre || 0);
  }
  return agg;
}

const AGE = {1:'0-4 ans', 2:'5-11 ans', 3:'12-17 ans', 4:'18-21 ans', 5:'22-44 ans', 6:'45-64 ans', 7:'65-74 ans', 8:'75 ans+', 9:'Total'};
function v(data, l, c) { return Math.round(data[`${l}-${c}`] || 0); }
function fmt(n) { return n.toLocaleString('fr-CA'); }

const d16 = getData(rawP16, anneeSelect, etabSelect);
const d17 = getData(rawP17, anneeSelect, etabSelect);
```

## Évolution de la liste d'attente — Toutes années

```js
// Build time series for selected etab
const allYears = meta.annees;
const trendData = [];

for (const annee of allYears) {
  const dd = getData(rawP17, annee, etabSelect);
  const total = v(dd, 3, 9);
  const di = v(dd, 1, 9);
  const tsa = v(dd, 2, 9);
  if (total > 0) {
    trendData.push({ annee, total, di, tsa });
  }
}

Plot.plot({
  title: `Évolution liste d'attente — ${etabSelect}`,
  x: { label: "Année", tickRotate: -30 },
  y: { label: "Personnes en attente", grid: true },
  color: { legend: true },
  marks: [
    Plot.line(trendData, {x: "annee", y: "di", stroke: "#378ADD", strokeWidth: 2, marker: "circle", label: "DI"}),
    Plot.line(trendData, {x: "annee", y: "tsa", stroke: "#1D9E75", strokeWidth: 2, marker: "circle", label: "TSA"}),
    Plot.line(trendData, {x: "annee", y: "total", stroke: "#D85A30", strokeWidth: 2.5, strokeDasharray: "4,3", marker: "circle", label: "Total"}),
    Plot.ruleY([0]),
  ],
  width: 700,
  height: 320,
})
```
