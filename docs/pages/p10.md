```js
function getData(annee, etab) {
  let d = rawData.filter(r => r.Annee === annee);
  if (etab !== "Tout le Québec") d = d.filter(r => r.NomEtablissement === etab);
  const agg = {};
  for (const r of d) {
    const key = `${r.Ligne}-${r.Colonne}`;
    agg[key] = (agg[key] || 0) + (r.Chiffre || 0);
  }
  return agg;
}

const AGE = {1:'0-4 ans', 2:'5-11 ans', 3:'12-17 ans', 4:'18-21 ans', 5:'22-44 ans', 6:'45-64 ans', 7:'65-74 ans', 8:'75 ans+', 9:'Total'};
function v(data, l, c) { return Math.round(data[`${l}-${c}`] || 0); }
function fmt(n) { return n.toLocaleString('fr-CA'); }
function pct(n, tot) { return tot > 0 ? (n/tot*100).toFixed(1) + "%" : "—"; }

const data = getData(anneeSelect, etabSelect);
```

## Lieu de résidence des usagers desservis

```js
const LIEUX = [
  { ligne: 1, label: "Milieu naturel (parents)" },
  { ligne: 2, label: "Milieu naturel (autonome)" },
  { ligne: 3, label: "Ressource non institutionnelle (RNI)" },
  { ligne: 4, label: "Autre RNI gérée par le CR" },
  { ligne: 5, label: "RRAC — avec assistance continue" },
  { ligne: 6, label: "RRAC — avec allocation" },
  { ligne: 7, label: "Admis dans un autre établissement" },
  { ligne: 8, label: "Ressource intermédiaire (RI)" },
  { ligne: 9, label: "Ressource de type familial (RTF)" },
  { ligne: 10, label: "Internat" },
  { ligne: 11, label: "Foyer de groupe" },
  { ligne: 12, label: "Autres" },
  { ligne: 13, label: "TOTAL (L.1 à L.12)" },
];

const total = v(data, 13, 9);

const tableData = LIEUX.map(r => ({
  "Lieu de résidence": r.label,
  ...Object.fromEntries([1,2,3,4,5,6,7,8].map(c => [AGE[c], fmt(v(data, r.ligne, c))])),
  "Total": fmt(v(data, r.ligne, 9)),
  "% du total": r.ligne === 13 ? "100%" : pct(v(data, r.ligne, 9), total),
}));

view(tableWithTools(Inputs.table(tableData, {width: "100%"}), tableData, "p10_lieux.csv"));
```
